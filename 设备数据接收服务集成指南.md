### 步骤3：修改现有的Netty Handler

修改 `RadarSystem.Communication/Handlers/RadarServerHandler.cs`（或创建新的）：

```csharp
using DotNetty.Transport.Channels;
using RadarSystem.Communication.Models;
using RadarSystem.Communication.Services;
using RadarSystem.Communication.Utilities;

namespace RadarSystem.Communication.Handlers
{
    /// <summary>
    /// 雷达服务器Handler - 接收设备数据并转发到处理服务
    /// </summary>
    public class ArcRadarServerHandler : SimpleChannelInboundHandler<byte[]>
    {
        private readonly RadarDataProcessorService _processorService;
        private readonly RadarDataValidator _validator;
        private readonly ILogger<ArcRadarServerHandler> _logger;
        
        // 设备映射表（SlaveId -> DeviceId）
        private readonly ConcurrentDictionary<string, string> _deviceIdMap 
            = new ConcurrentDictionary<string, string>();

        public ArcRadarServerHandler(
            RadarDataProcessorService processorService,
            RadarDataValidator validator,
            ILogger<ArcRadarServerHandler> logger)
        {
            _processorService = processorService;
            _validator = validator;
            _logger = logger;
            
            // 初始化时加载设备映射
            LoadDeviceMapping();
        }

        protected override void ChannelRead0(IChannelHandlerContext ctx, byte[] msg)
        {
            try
            {
                // 1. 验证数据包格式
                if (!_validator.ValidatePacketFormat(msg))
                {
                    _logger.LogWarning("无效的数据包格式，已丢弃");
                    return;
                }
                
                // 2. 解析数据包
                var packet = ParseDataPacket(msg, ctx);
                
                // 3. 转发到处理服务（不验证设备是否配置）
                _processorService.ReceiveData(packet);
                
                _logger.LogDebug("数据包已转发到处理服务: {Packet}", packet);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "处理接收数据时发生异常");
            }
        }

        /// <summary>
        /// 解析数据包
        /// </summary>
        private RadarDataPacket ParseDataPacket(byte[] data, IChannelHandlerContext ctx)
        {
            // 转换为16进制字符串
            string hexString = BitConverter.ToString(data).Replace("-", "");
            
            // 解析命令码（第13-16位）
            string command = hexString.Substring(12, 4);
            
            // 解析SlaveId（第5-12位）
            string slaveIdHex = hexString.Substring(4, 8);
            string slaveId = Convert.ToInt32(slaveIdHex, 16).ToString();
            
            // 尝试从映射表获取DeviceId（如果没有也不影响）
            string? deviceId = _deviceIdMap.TryGetValue(slaveId, out string? value) 
                ? value 
                : null;
            
            // 确定数据类型
            string dataType = command switch
            {
                "0302" => "00",  // 形变数据
                "0301" => "01",  // 复散射数据
                "0303" => "02",  // 置信度数据
                _ => "99"        // 未知类型
            };
            
            // 获取远程地址
            string remoteAddress = ctx.Channel.RemoteAddress?.ToString() ?? "Unknown";
            
            return new RadarDataPacket
            {
                SlaveId = slaveId,
                DeviceId = deviceId,
                ProjectId = "DEFAULT",  // 可以从配置或数据库读取
                Command = command,
                DataType = dataType,
                RawData = data,
                ReceiveTime = DateTime.Now,
                RemoteAddress = remoteAddress
            };
        }

        /// <summary>
        /// 加载设备映射表（可选）
        /// 注意：即使没有映射也不影响数据接收和保存
        /// </summary>
        private async void LoadDeviceMapping()
        {
            try
            {
                // 这里可以从数据库或API加载设备映射
                // 示例：从API获取
                // var devices = await _apiClient.GetDevicesAsync();
                // foreach (var device in devices)
                // {
                //     _deviceIdMap[device.SlaveId] = device.DeviceId;
                // }
                
                _logger.LogInformation("设备映射加载完成，共 {Count} 个设备", _deviceIdMap.Count);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "加载设备映射失败");
                // 不抛出异常，允许系统继续运行
            }
        }

        public override void ChannelActive(IChannelHandlerContext context)
        {
            _logger.LogInformation("连接已建立: {RemoteAddress}", context.Channel.RemoteAddress);
            base.ChannelActive(context);
        }

        public override void ChannelInactive(IChannelHandlerContext context)
        {
            _logger.LogInformation("连接已断开: {RemoteAddress}", context.Channel.RemoteAddress);
            base.ChannelInactive(context);
        }

        public override void ExceptionCaught(IChannelHandlerContext context, Exception exception)
        {
            _logger.LogError(exception, "通道异常");
            context.CloseAsync();
        }
    }
}
```

### 步骤4：添加监控API（可选）

创建 `RadarSystem.WebAPI/Controllers/RadarDataMonitorController.cs`：

```csharp
using Microsoft.AspNetCore.Mvc;
using RadarSystem.Communication.Services;

namespace RadarSystem.WebAPI.Controllers
{
    [ApiController]
    [Route("api/radar-data-monitor")]
    public class RadarDataMonitorController : ControllerBase
    {
        private readonly RadarDataProcessorService _processorService;

        public RadarDataMonitorController(RadarDataProcessorService processorService)
        {
            _processorService = processorService;
        }

        /// <summary>
        /// 获取数据处理统计信息
        /// GET /api/radar-data-monitor/statistics
        /// </summary>
        [HttpGet("statistics")]
        public IActionResult GetStatistics()
        {
            var (received, processed, md5Failed, saveFailed) = _processorService.GetStatistics();
            int queueSize = _processorService.GetQueueSize();

            return Ok(new
            {
                code = 200,
                data = new
                {
                    totalReceived = received,
                    totalProcessed = processed,
                    md5Failed = md5Failed,
                    saveFailed = saveFailed,
                    queueSize = queueSize,
                    successRate = received > 0 ? (double)processed / received * 100 : 0
                }
            });
        }

        /// <summary>
        /// 获取当前队列大小
        /// GET /api/radar-data-monitor/queue-size
        /// </summary>
        [HttpGet("queue-size")]
        public IActionResult GetQueueSize()
        {
            return Ok(new
            {
                code = 200,
                data = new
                {
                    queueSize = _processorService.GetQueueSize()
                }
            });
        }
    }
}
```

## 🐛 常见问题

### Q1: 数据没有被处理

**检查**：
1. 查看日志是否有 "雷达数据处理服务已启动"
2. 检查队列大小：`GET /api/radar-data-monitor/queue-size`
3. 确认Handler正确调用了 `_processorService.ReceiveData(packet)`

### Q2: MD5校验总是失败

**检查**：
1. 确认数据包格式是否正确（协议头5A5A或3C3C）
2. 检查MD5位置是否正确（offset+4 到 offset+20）
3. 临时禁用MD5校验进行测试：`"EnableMD5Check": false`

### Q3: 文件保存失败

**检查**：
1. 确认DataPath目录有写权限
2. 检查磁盘空间是否充足
3. 查看日志中的具体错误信息

### Q4: 没有设备ID怎么办？

**答案**：
- 系统会自动使用SlaveId作为设备标识
- 文件路径会变成：`/data/project/DEFAULT/radar/{SlaveId}/...`
- 这完全符合需求，因为不需要验证设备配置
