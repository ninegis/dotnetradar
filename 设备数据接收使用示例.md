### 示例1：在Netty Handler中接收数据

```csharp
using DotNetty.Transport.Channels;
using RadarSystem.Communication.Models;
using RadarSystem.Communication.Services;
using RadarSystem.Communication.Utilities;

namespace RadarSystem.Communication.Handlers
{
    /// <summary>
    /// 圆弧雷达Handler - 接收数据并转发到处理服务
    /// </summary>
    public class ArcRadarHandler : SimpleChannelInboundHandler<byte[]>
    {
        private readonly RadarDataProcessorService _processorService;
        private readonly RadarDataValidator _validator;
        private readonly ILogger<ArcRadarHandler> _logger;

        public ArcRadarHandler(
            RadarDataProcessorService processorService,
            RadarDataValidator validator,
            ILogger<ArcRadarHandler> logger)
        {
            _processorService = processorService;
            _validator = validator;
            _logger = logger;
        }

        protected override void ChannelRead0(IChannelHandlerContext ctx, byte[] msg)
        {
            try
            {
                _logger.LogDebug("接收到数据包，长度: {Length} bytes", msg.Length);
                
                // 1. 快速验证格式
                if (!_validator.ValidatePacketFormat(msg))
                {
                    _logger.LogWarning("数据包格式无效，已丢弃");
                    return;
                }
                
                // 2. 解析数据包
                var packet = ParsePacket(msg, ctx);
                
                // 3. 发送到处理服务（不验证设备配置）
                bool success = _processorService.ReceiveData(packet);
                
                if (!success)
                {
                    _logger.LogWarning("数据队列已满，数据被丢弃: {Packet}", packet);
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "处理数据接收时发生异常");
            }
        }

        /// <summary>
        /// 解析数据包
        /// </summary>
        private RadarDataPacket ParsePacket(byte[] data, IChannelHandlerContext ctx)
        {
            // 转换为16进制字符串便于解析
            string hexString = BitConverter.ToString(data).Replace("-", "");
            
            // 解析命令码（位置12-16）
            string command = hexString.Substring(12, 4);
            
            // 解析SlaveId（位置4-12）
            string slaveIdHex = hexString.Substring(4, 8);
            int slaveIdInt = Convert.ToInt32(slaveIdHex, 16);
            string slaveId = slaveIdInt.ToString();
            
            // 确定数据类型
            string dataType = command switch
            {
                "0302" => "00",  // 形变
                "0301" => "01",  // 复散射
                "0303" => "02",  // 置信度
                _ => "99"
            };
            
            // 获取远程地址
            string remoteAddress = ctx.Channel.RemoteAddress?.ToString() ?? "Unknown";
            
            _logger.LogDebug("解析数据包: SlaveId={SlaveId}, Command={Command}, DataType={DataType}", 
                slaveId, command, dataType);
            
            return new RadarDataPacket
            {
                SlaveId = slaveId,
                DeviceId = null,  // 不验证设备配置，可以为null
                ProjectId = "DEFAULT",
                Command = command,
                DataType = dataType,
                RawData = data,
                ReceiveTime = DateTime.Now,
                RemoteAddress = remoteAddress
            };
        }
    }
}
```

### 示例3：监控数据处理状态

```csharp
using Microsoft.AspNetCore.Mvc;
using RadarSystem.Communication.Services;

namespace RadarSystem.WebAPI.Controllers
{
    [ApiController]
    [Route("api/radar-data-monitor")]
    public class RadarDataMonitorController : ControllerBase
    {
        private readonly RadarDataProcessorService _processorService;
        private readonly ILogger<RadarDataMonitorController> _logger;

        public RadarDataMonitorController(
            RadarDataProcessorService processorService,
            ILogger<RadarDataMonitorController> logger)
        {
            _processorService = processorService;
            _logger = logger;
        }

        /// <summary>
        /// 获取实时统计信息
        /// GET /api/radar-data-monitor/statistics
        /// </summary>
        [HttpGet("statistics")]
        public IActionResult GetStatistics()
        {
            try
            {
                var (received, processed, md5Failed, saveFailed) = _processorService.GetStatistics();
                int queueSize = _processorService.GetQueueSize();

                double successRate = received > 0 
                    ? Math.Round((double)processed / received * 100, 2) 
                    : 0;

                return Ok(new
                {
                    code = 200,
                    message = "获取统计信息成功",
                    data = new
                    {
                        totalReceived = received,
                        totalProcessed = processed,
                        md5Failed = md5Failed,
                        saveFailed = saveFailed,
                        currentQueueSize = queueSize,
                        successRate = successRate,
                        status = queueSize > 5000 ? "队列积压" : "正常",
                        updateTime = DateTime.Now
                    }
                });
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "获取统计信息失败");
                return Ok(new { code = 500, message = $"获取失败: {ex.Message}" });
            }
        }

        /// <summary>
        /// 获取当前队列大小
        /// GET /api/radar-data-monitor/queue-size
        /// </summary>
        [HttpGet("queue-size")]
        public IActionResult GetQueueSize()
        {
            try
            {
                int queueSize = _processorService.GetQueueSize();
                
                return Ok(new
                {
                    code = 200,
                    data = new
                    {
                        queueSize = queueSize,
                        status = queueSize > 8000 
                            ? "严重积压" 
                            : queueSize > 5000 
                                ? "轻度积压" 
                                : "正常",
                        maxSize = 10000,
                        usage = Math.Round((double)queueSize / 10000 * 100, 2)
                    }
                });
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "获取队列大小失败");
                return Ok(new { code = 500, message = $"获取失败: {ex.Message}" });
            }
        }

        /// <summary>
        /// 健康检查
        /// GET /api/radar-data-monitor/health
        /// </summary>
        [HttpGet("health")]
        public IActionResult HealthCheck()
        {
            try
            {
                var (received, processed, md5Failed, saveFailed) = _processorService.GetStatistics();
                int queueSize = _processorService.GetQueueSize();

                bool isHealthy = queueSize < 8000 && saveFailed < received * 0.01;

                return Ok(new
                {
                    code = 200,
                    data = new
                    {
                        healthy = isHealthy,
                        queueSize = queueSize,
                        errorRate = received > 0 
                            ? Math.Round((double)(md5Failed + saveFailed) / received * 100, 2) 
                            : 0,
                        message = isHealthy ? "系统运行正常" : "系统存在异常"
                    }
                });
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "健康检查失败");
                return Ok(new
                {
                    code = 500,
                    data = new
                    {
                        healthy = false,
                        message = $"健康检查失败: {ex.Message}"
                    }
                });
            }
        }
    }
}
```

### 示例4：查询设备数据文件

```csharp
using Microsoft.AspNetCore.Mvc;
using RadarSystem.Communication.Interfaces;

namespace RadarSystem.WebAPI.Controllers
{
    [ApiController]
    [Route("api/radar-data-files")]
    public class RadarDataFilesController : ControllerBase
    {
        private readonly IRadarFileStorage _fileStorage;
        private readonly ILogger<RadarDataFilesController> _logger;

        public RadarDataFilesController(
            IRadarFileStorage fileStorage,
            ILogger<RadarDataFilesController> logger)
        {
            _fileStorage = fileStorage;
            _logger = logger;
        }

        /// <summary>
        /// 获取设备的数据文件列表
        /// GET /api/radar-data-files/{projectId}/{deviceId}?date=20250426
        /// </summary>
        [HttpGet("{projectId}/{deviceId}")]
        public async Task<IActionResult> GetDeviceDataFiles(
            string projectId, 
            string deviceId, 
            [FromQuery] string? date = null)
        {
            try
            {
                _logger.LogInformation("查询设备数据文件: ProjectId={ProjectId}, DeviceId={DeviceId}, Date={Date}", 
                    projectId, deviceId, date);

                var files = await _fileStorage.GetDeviceDataFilesAsync(projectId, deviceId, date);

                return Ok(new
                {
                    code = 200,
                    message = "查询成功",
                    data = new
                    {
                        projectId = projectId,
                        deviceId = deviceId,
                        date = date,
                        fileCount = files.Length,
                        files = files.Select(f => new
                        {
                            filePath = f,
                            fileName = Path.GetFileName(f),
                            size = new FileInfo(f).Length,
                            createTime = File.GetCreationTime(f),
                            dataType = Path.GetFileName(f).StartsWith("X") ? "形变" :
                                      Path.GetFileName(f).StartsWith("F") ? "复散射" :
                                      Path.GetFileName(f).StartsWith("Z") ? "置信度" : "未知"
                        }).ToArray()
                    }
                });
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "查询设备数据文件失败");
                return Ok(new { code = 500, message = $"查询失败: {ex.Message}" });
            }
        }

        /// <summary>
        /// 读取数据文件内容
        /// GET /api/radar-data-files/content?filePath={filePath}
        /// </summary>
        [HttpGet("content")]
        public async Task<IActionResult> GetFileContent([FromQuery] string filePath)
        {
            try
            {
                if (!_fileStorage.FileExists(filePath))
                {
                    return Ok(new { code = 404, message = "文件不存在" });
                }

                byte[] data = await _fileStorage.ReadRadarDataAsync(filePath);

                return File(data, "application/octet-stream", Path.GetFileName(filePath));
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "读取文件失败: {FilePath}", filePath);
                return Ok(new { code = 500, message = $"读取失败: {ex.Message}" });
            }
        }
    }
}
```

```csharp
using Xunit;
using Microsoft.Extensions.Logging;
using RadarSystem.Communication.Models;
using RadarSystem.Communication.Utilities;

namespace RadarSystem.Tests
{
    public class RadarDataValidatorTests
    {
        private readonly ILogger<RadarDataValidator> _logger;
        private readonly RadarDataValidator _validator;

        public RadarDataValidatorTests()
        {
            var loggerFactory = LoggerFactory.Create(builder => builder.AddConsole());
            _logger = loggerFactory.CreateLogger<RadarDataValidator>();
            _validator = new RadarDataValidator(_logger);
        }

        [Fact]
        public void ValidatePacketFormat_ValidHeader_ReturnsTrue()
        {
            // Arrange
            byte[] data = new byte[100];
            data[0] = 0x5A;
            data[1] = 0x5A;

            // Act
            bool result = _validator.ValidatePacketFormat(data);

            // Assert
            Assert.True(result);
        }

        [Fact]
        public void ValidatePacketFormat_InvalidHeader_ReturnsFalse()
        {
            // Arrange
            byte[] data = new byte[100];
            data[0] = 0xFF;
            data[1] = 0xFF;

            // Act
            bool result = _validator.ValidatePacketFormat(data);

            // Assert
            Assert.False(result);
        }

        [Fact]
        public void ValidateMD5_ValidData_ReturnsTrue()
        {
            // Arrange
            // 创建测试数据包（包含正确的MD5）
            byte[] testData = CreateTestPacketWithMD5();
            var packet = new RadarDataPacket
            {
                SlaveId = "120",
                RawData = testData
            };

            // Act
            bool result = _validator.ValidateMD5(packet);

            // Assert
            Assert.True(result);
        }

        private byte[] CreateTestPacketWithMD5()
        {
            // 创建测试数据包
            // [Header(12)] [Offset(4)] [MD5(16)] [Data(100)]
            byte[] packet = new byte[132];
            
            // 设置协议头
            packet[0] = 0x5A;
            packet[1] = 0x5A;
            
            // 实际数据
            byte[] actualData = new byte[100];
            for (int i = 0; i < 100; i++)
            {
                actualData[i] = (byte)i;
            }
            
            // 计算MD5
            using (var md5 = System.Security.Cryptography.MD5.Create())
            {
                byte[] hash = md5.ComputeHash(actualData);
                Array.Copy(hash, 0, packet, 16, 16);  // 放在offset+4的位置
            }
            
            // 复制实际数据
            Array.Copy(actualData, 0, packet, 32, 100);  // 从offset+20开始
            
            return packet;
        }
    }
}
```

## 🎯 实际使用场景

### 场景1：新设备上线（无需预配置）

```
1. 新雷达设备连接到系统
2. 设备发送数据（SlaveId=150）
3. 系统解析SlaveId
4. 在映射表中找不到DeviceId → 使用SlaveId
5. 文件路径：/data/project/DEFAULT/radar/150/20250426/X{uuid}
6. 数据正常保存 ✅
7. MQTT通知发送 ✅
```

**结论**：✅ 即使设备未在数据库配置，数据也能正常保存

### 场景2：已配置设备

```
1. 已配置设备连接（SlaveId=120 → DeviceId=DEVICE001）
2. 系统解析SlaveId=120
3. 从映射表获取DeviceId=DEVICE001
4. 文件路径：/data/project/PROJECT001/radar/DEVICE001/20250426/X{uuid}
5. 数据保存 ✅
6. TDengine记录元数据 ✅
7. MQTT通知发送 ✅
```

**结论**：✅ 已配置设备使用更完整的信息存储

### 场景3：MD5校验失败

```
1. 接收到损坏的数据包
2. MD5校验失败
3. 记录日志：MD5Failed++
4. 丢弃数据包
5. 不影响后续数据接收
```

**结论**：✅ 损坏数据被过滤，不影响正常数据

### 场景4：队列积压

```
1. 数据接收速度 > 处理速度
2. 队列大小增长到8000
3. 监控API返回"严重积压"状态
4. 管理员收到告警
5. 调整ProcessThreadCount配置
```

**结论**：✅ 可监控、可调整
